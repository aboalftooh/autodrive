enum class SettingsRowVariant { Value, Navigation, Editable, Destructive, Status }

data class SettingsGroupItem(
    val id: String,
    val label: String,
    val value: String? = null,
    val variant: SettingsRowVariant = SettingsRowVariant.Value,
    val enabled: Boolean = true,
)

@Composable
fun SettingsRow(
    label: String,
    modifier: Modifier = Modifier,
    value: String? = null,
    variant: SettingsRowVariant = SettingsRowVariant.Value,
    enabled: Boolean = true,
    statusTone: AutoDriveStatusTone = AutoDriveStatusTone.Neutral,
    onClick: (() -> Unit)? = null,
) {
    val titleColor = if (variant == SettingsRowVariant.Destructive) AutoDriveStatus.Error else if (enabled) AutoDriveText.Primary else AutoDriveText.Disabled
    AutoDriveListRow(
        title = label,
        supportingText = null,
        enabled = enabled,
        onClick = onClick,
        modifier = modifier,
        trailing = {
            when (variant) {
                SettingsRowVariant.Navigation, SettingsRowVariant.Editable -> Icon(Icons.AutoMirrored.Rounded.KeyboardArrowLeft, null, tint = AutoDriveText.Secondary, modifier = Modifier.then(Modifier))
                SettingsRowVariant.Status -> if (value != null) AutoDriveStatusChip(value, statusTone)
                else -> if (value != null) Text(value, style = MaterialTheme.typography.bodyMedium, color = AutoDriveText.Secondary)
            }
        },
    )
}

@Composable
fun SettingsGroup(
    title: String,
    items: List<SettingsGroupItem>,
    modifier: Modifier = Modifier,
    onItemClick: (String) -> Unit = {},
    headerAction: (@Composable () -> Unit)? = null,
) {
    Column(modifier) {
        AutoDriveSectionHeader(title, action = headerAction)
        AutoDriveCard(Modifier.fillMaxWidth()) {
            items.forEachIndexed { index, item ->
                SettingsRow(item.label, value = item.value, variant = item.variant, enabled = item.enabled, onClick = { onItemClick(item.id) })
                if (index != items.lastIndex) AutoDriveDivider()
            }
        }
    }
}
